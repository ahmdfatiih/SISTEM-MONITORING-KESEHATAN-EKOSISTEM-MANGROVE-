PANEL WEBGIS var panel = ui.Panel({ style: { width: '350px', padding: '12px' } }); panel.add(ui.Label('WEBGIS EKOSISTEM MANGROVE', { fontSize: '20px', fontWeight: 'bold' })); panel.add(ui.Label( 'Desa Sidodadi, Kabupaten Pesawaran, Lampung', {fontSize: '12px'} )); panel.add(ui.Label('Pilih tahun pengamatan:', { fontWeight: 'bold', margin: '18px 0 5px 0' })); var pilihanTahun = ui.Select({ items: ['2021', '2022', '2023', '2024', '2025', '2026'], value: '2021', style: {stretch: 'horizontal'} }); panel.add(pilihanTahun); var status = ui.Label('Memuat raster...', { color: 'blue', margin: '8px 0' }); panel.add(status); panel.add(ui.Label('Indeks Kesehatan Mangrove', { fontWeight: 'bold', margin: '15px 0 5px 0' })); var tabel = ui.Panel(); panel.add(tabel); // 6.
